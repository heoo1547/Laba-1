import math

def calculate_bmi(weight, height):
    """Индекс массы тела: вес (кг) / рост (м)^2"""
    return weight / (height ** 2)

def solve_quadratic(a, b, c):
    """Решение квадратного уравнения ax^2 + bx + c = 0"""
    discriminant = b**2 - 4*a*c
    if discriminant < 0:
        return "Нет действительных корней"
    elif discriminant == 0:
        x = -b / (2*a)
        return f"Один корень: x = {x}"
    else:
        x1 = (-b + math.sqrt(discriminant)) / (2*a)
        x2 = (-b - math.sqrt(discriminant)) / (2*a)
        return f"Два корня: x1 = {x1}, x2 = {x2}"

def main():
    print("="*50)
    print("        Многофункциональный калькулятор")
    print("="*50)
    print("Выберите операцию:")
    print("1. Сложение (+)")
    print("2. Вычитание (-)")
    print("3. Умножение (*)")
    print("4. Деление (/)")
    print("5. Целочисленное деление (//)")
    print("6. Остаток от деления (%)")
    print("7. Возведение в степень (**)")
    print("8. Квадратный корень (√)")
    print("9. Кубический корень (∛)")
    print("10. Факториал числа (n!)")
    print("11. Индекс массы тела (ИМТ)")
    print("12. Квадратное уравнение")
    print("13. Проценты (X% от Y)")
    print("14. Перевод температур (C <-> F <-> K)")
    print("0. Выход")

    while True:
        try:
            choice = input("\nВведите номер операции (0-14): ").strip()

            if choice == '0':
                print("Выход из программы. До свидания!")
                break

            # Операции с двумя числами
            elif choice in ['1', '2', '3', '4', '5', '6', '7']:
                a = float(input("Введите первое число: "))
                b = float(input("Введите второе число: "))

                if choice == '1':
                    print(f"Результат: {a} + {b} = {a + b}")
                elif choice == '2':
                    print(f"Результат: {a} - {b} = {a - b}")
                elif choice == '3':
                    print(f"Результат: {a} * {b} = {a * b}")
                elif choice == '4':
                    if b == 0:
                        print("Ошибка: Деление на ноль!")
                    else:
                        print(f"Результат: {a} / {b} = {a / b}")
                elif choice == '5':
                    if b == 0:
                        print("Ошибка: Деление на ноль!")
                    else:
                        print(f"Результат: {a} // {b} = {a // b}")
                elif choice == '6':
                    if b == 0:
                        print("Ошибка: Деление на ноль!")
                    else:
                        print(f"Результат: {a} % {b} = {a % b}")
                elif choice == '7':
                    print(f"Результат: {a} ** {b} = {a ** b}")

            # Корни (одно число)
            elif choice == '8':
                num = float(input("Введите число для извлечения квадратного корня: "))
                if num < 0:
                    print("Ошибка: Нельзя извлечь квадратный корень из отрицательного числа (комплексные числа не поддерживаются).")
                else:
                    print(f"Квадратный корень из {num} = {math.sqrt(num)}")
            elif choice == '9':
                num = float(input("Введите число для извлечения кубического корня: "))
                # Кубический корень можно через ** (1/3) или math.pow для отрицательных
                if num >= 0:
                    result = num ** (1/3)
                else:
                    result = -(-num) ** (1/3)
                print(f"Кубический корень из {num} = {result}")

            # Факториал
            elif choice == '10':
                num = int(input("Введите целое неотрицательное число: "))
                if num < 0:
                    print("Ошибка: Факториал отрицательного числа не определен.")
                else:
                    print(f"{num}! = {math.factorial(num)}")

            # ИМТ
            elif choice == '11':
                weight = float(input("Введите вес (в кг): "))
                height = float(input("Введите рост (в метрах, например 1.75): "))
                if height <= 0:
                    print("Ошибка: Рост должен быть положительным числом.")
                else:
                    bmi = calculate_bmi(weight, height)
                    print(f"Ваш ИМТ = {bmi:.2f}")
                    # Интерпретация результата
                    if bmi < 18.5:
                        print("Интерпретация: Дефицит массы тела")
                    elif 18.5 <= bmi < 25:
                        print("Интерпретация: Норма")
                    elif 25 <= bmi < 30:
                        print("Интерпретация: Избыточная масса тела (предожирение)")
                    else:
                        print("Интерпретация: Ожирение")

            # Квадратное уравнение
            elif choice == '12':
                print("Решение уравнения ax^2 + bx + c = 0")
                a = float(input("Введите a: "))
                b = float(input("Введите b: "))
                c = float(input("Введите c: "))
                if a == 0:
                    print("Это не квадратное уравнение (a не должно быть равно 0).")
                else:
                    print(solve_quadratic(a, b, c))

            # Проценты
            elif choice == '13':
                percent = float(input("Сколько процентов? "))
                number = float(input("От какого числа? "))
                result = (percent / 100) * number
                print(f"{percent}% от {number} = {result}")

            # Перевод температур
            elif choice == '14':
                print("Перевод температур:")
                print("1. Цельсий -> Фаренгейт")
                print("2. Фаренгейт -> Цельсий")
                print("3. Цельсий -> Кельвин")
                print("4. Кельвин -> Цельсий")
                temp_choice = input("Выберите направление (1-4): ")

                if temp_choice == '1':
                    c = float(input("Введите температуру в Цельсиях: "))
                    f = (c * 9/5) + 32
                    print(f"{c}°C = {f}°F")
                elif temp_choice == '2':
                    f = float(input("Введите температуру в Фаренгейтах: "))
                    c = (f - 32) * 5/9
                    print(f"{f}°F = {c}°C")
                elif temp_choice == '3':
                    c = float(input("Введите температуру в Цельсиях: "))
                    k = c + 273.15
                    print(f"{c}°C = {k}K")
                elif temp_choice == '4':
                    k = float(input("Введите температуру в Кельвинах: "))
                    if k < 0:
                        print("Ошибка: Температура в Кельвинах не может быть отрицательной.")
                    else:
                        c = k - 273.15
                        print(f"{k}K = {c}°C")
                else:
                    print("Неверный выбор направления.")

            else:
                print("Неверный ввод. Пожалуйста, выберите номер от 0 до 14.")

        except ValueError:
            print("Ошибка: Введите число корректно (используйте точку для дробных).")
        except Exception as e:
            print(f"Произошла ошибка: {e}")

if __name__ == "__main__":
    main()
